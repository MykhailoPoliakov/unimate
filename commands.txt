seed:
    cd backend; python -m app.seed
    (wipes backend/unimate.db including news and users, then seeds)

launch backend:
    cd backend; python -m uvicorn app.main:app --reload --host 0.0.0.0 --port 8000
    
    backend url:
        http://127.0.0.1:8000

open database:
    cd backend; python -m sqlite3 .\unimate.db

    add admin:
        python -m app.admin grant USER_ID_HERE

    remove admin:
        python -m app.admin revoke USER_ID_HERE

    list users:
        python -m app.admin list
        

launch frontend:
    cd frontend; npx expo start
